Shoppable Cart

Documentación de uso, configuración y personalización del script shoppable-cart.js

Descripción Instalación Productos Variantes Ubicación Estilos 7. Script Completo

1. ¿Qué hace este script?

El script crea una interfaz de productos y un carrito directamente desde JavaScript. Al inicializarse genera el botón “🛒 Ver productos”, una mini ventana centrada para mostrar los productos y el carrito, y una ventana modal para consultar los detalles de cada producto.

El catálogo se define dentro del propio archivo mediante el arreglo P. La moneda está configurada inicialmente como COP.

El archivo funciona como un componente autocontenido: agrega sus propios estilos, crea los elementos HTML necesarios y administra el estado del carrito en memoria. El botón “Ver productos” puede colocarse dentro del elemento indicado mediante data-target.

2. Instalación básica

Guarda shoppable-cart.js en tu proyecto y cárgalo desde el HTML:

<script src="https://pawangang0329.github.io/Script_Shoppable/shoppable-cart.js"></script>

También puedes indicar un elemento específico donde se insertará el botón usando data-target:

<div id="mi-carrito"></div>

<script src="https://pawangang0329.github.io/Script_Shoppable/shoppable-cart.js"
        data-target="#mi-carrito"></script>

Si no se especifica data-target, el script utiliza document.body como contenedor del botón. La mini ventana de productos y carrito se crea en el documento para mostrarse centrada sobre la página.

3. Cómo agregar o modificar productos

Los productos se encuentran en el arreglo P. Cada producto utiliza esta estructura:

{
  id: "1",
  name: "Nombre_producto_1",
  description: "Descripcion_producto_1",
  image: "URL_imagen_producto_1",
  url: "URL_oficial_producto_1",
  variants: [
    {
      id: "v1",
      name: "Variante_1",
      image: "URL_imagen_variante_1",
      price: 10000
    }
  ],
  specifications: [
    "Caracteristica_1",
    "Caracteristica_2"
  ]

4. Variantes

Cada variante tiene su propio id, nombre, imagen y precio. El usuario puede seleccionar la variante desde la ventana de detalle del producto.

El carrito identifica un artículo mediante la combinación de productId y variantId. Por eso, dos variantes del mismo producto pueden mantenerse como artículos separados.

Importante: cada variante puede tener una imagen propia y esa imagen se guarda al agregar el artículo al carrito; sin embargo, el modal de detalle no cambia automáticamente la imagen principal cuando se selecciona otra variante.

5. Elegir dónde aparece el componente

El atributo data-target determina dónde se coloca el botón “🛒 Ver productos”. Por ejemplo, si quieres que aparezca debajo de un video y alineado a la izquierda, puedes crear una zona específica: 

<section id="zona-productos"></section>

<script
  src="https://pawangang0329.github.io/Script_Shoppable/shoppable-cart.js"
  data-target="#zona-productos">
</script>

El selector debe coincidir con un elemento existente en la página. La mini ventana del catálogo/carrito no se convierte en una barra lateral: se muestra centrada sobre la página cuando el usuario pulsa el botón.

Importante: data-target controla principalmente la ubicación del botón. La ventana de productos y carrito se genera aparte para poder mostrarse como una mini ventana centrada.

6. Personalizar apariencia

Los estilos se generan automáticamente dentro de la función css(). La versión actual utiliza una mini ventana centrada en lugar de un panel lateral.

ClaseElemento

.sppMini ventana centrada que contiene productos y carrito.
.spheaderEncabezado rojo con el título “🛒 Productos”.
.spcart-closeBotón X situado a la derecha del título para cerrar toda la ventana.
.spcardTarjeta individual de cada producto.
.spqControles de cantidad − / +.
.moreBotón rojo y ancho “Ver más”.
.spiFila de un producto dentro del carrito.
.spbuyBotón rojo “PAGAR TODO”.
.spemptyMensaje visual cuando el carrito está vacío.
.spmVentana modal de detalle de “Ver más”.
.addBotón “Agregar al carrito”.
.buyBotón rojo “Comprar ahora”.
.close-productBotón para cerrar la ventana de detalle.
.spoverlayFondo semitransparente que aparece detrás de la mini ventana.
.spresizeCuatro zonas invisibles ubicadas en las esquinas para modificar manualmente el tamaño de toda la ventana.

Cambiar el tamaño de la mini ventana

El ancho actual está definido en .spp:

.spp{
  width:505px;
}

Cambiar el color rojo principal

El rojo utilizado actualmente es #CE191E. Puedes reemplazarlo por otro color en las reglas que utilizan background:#CE191E. 

.spbuy{
  background:#CE191E;
}

Cambiar el color de los botones + y −

Los controles de cantidad del catálogo y carrito utilizan actualmente un tono amarillo: 

.spq button{
  background:#F5C542;
}
En la ventana de detalle del producto, los botones + y − permanecen blancos, mientras que los controles del catálogo y del carrito utilizan el tono amarillo.

    return cart.reduce(function(a,i){

  function add(p,v,n){
    var i = cart.find(function(x){
      return x.productId === p.id && x.variantId === v.id;
    });

    if(i){
      i.quantity += n;
    }else{
      cart.push({
        productId:p.id,
        variantId:v.id,
        name:p.name,
        variant:v.name,
        image:v.image || p.image,
        price:v.price,
        quantity:n,
        url:p.url
      });
    }

    renderCart();
  }

  function change(p,v,n){
    var i = cart.find(function(x){
      return x.productId === p && x.variantId === v;
    });

    if(!i)return;

    i.quantity += n;

    if(i.quantity < 1){
      cart = cart.filter(function(x){
        return x !== i;
      });
    }

    renderCart();
  }

  function remove(p,v){
    cart = cart.filter(function(i){
      return !(i.productId === p && i.variantId === v);
    });

    renderCart();
  }

  function checkout(){
    if(!cart.length)return;

    cart.forEach(function(i){
      if(i.url){
        window.open(i.url,"_blank");

    });

  function makeDraggable(el){
    var header=el.querySelector(".spheader");
    var dragging=false,startX=0,startY=0,startLeft=0,startTop=0;
    header.addEventListener("mousedown",function(e){
      if(e.target.closest(".spcart-close") || e.target.closest(".spresize")) return;
      var rect=el.getBoundingClientRect();
      el.style.transform="none";
      el.style.left=rect.left+"px";
      el.style.top=rect.top+"px";
      startX=e.clientX; startY=e.clientY; startLeft=rect.left; startTop=rect.top; dragging=true;
      document.body.style.userSelect="none"; e.preventDefault();
    });
    document.addEventListener("mousemove",function(e){
      if(!dragging)return;
      var rect=el.getBoundingClientRect();
      var left=Math.max(0,Math.min(startLeft+e.clientX-startX,window.innerWidth-rect.width));
      var top=Math.max(0,Math.min(startTop+e.clientY-startY,window.innerHeight-rect.height));
      el.style.left=left+"px"; el.style.top=top+"px";
    });
    document.addEventListener("mouseup",function(){if(dragging){dragging=false;document.body.style.userSelect="";}});
  }

  function makeResizable(el){
    el.querySelectorAll(".spresize").forEach(function(handle){
      handle.addEventListener("mousedown",function(e){
        e.preventDefault();e.stopPropagation();
        var rect=el.getBoundingClientRect();
        el.style.transform="none"; el.style.left=rect.left+"px"; el.style.top=rect.top+"px";
        var sx=e.clientX,sy=e.clientY,sw=rect.width,sh=rect.height,sl=rect.left,st=rect.top,type=handle.className;
        document.body.style.userSelect="none";
        function move(ev){
          var dx=ev.clientX-sx,dy=ev.clientY-sy,w=sw,h=sh,l=sl,t=st;
          var minW=window.innerWidth<600?300:330,minH=window.innerHeight<600?380:400;
          if(type.indexOf("tr")!==-1||type.indexOf("br")!==-1)w=sw+dx;
          if(type.indexOf("tl")!==-1||type.indexOf("bl")!==-1){w=sw-dx;l=sl+dx;}
          if(type.indexOf("bl")!==-1||type.indexOf("br")!==-1)h=sh+dy;
          if(type.indexOf("tl")!==-1||type.indexOf("tr")!==-1){h=sh-dy;t=st+dy;}
          if(w<minW){if(type.indexOf("tl")!==-1||type.indexOf("bl")!==-1)l=sl+sw-minW;w=minW;}
          if(h<minH){if(type.indexOf("tl")!==-1||type.indexOf("tr")!==-1)t=st+sh-minH;h=minH;}
          w=Math.min(w,window.innerWidth); h=Math.min(h,window.innerHeight);
          l=Math.max(0,Math.min(l,window.innerWidth-w)); t=Math.max(0,Math.min(t,window.innerHeight-h));
          el.style.width=w+"px";el.style.height=h+"px";el.style.left=l+"px";el.style.top=t+"px";
        }
        function stop(){document.removeEventListener("mousemove",move);document.removeEventListener("mouseup",stop);document.body.style.userSelect="";}
        document.addEventListener("mousemove",move);document.addEventListener("mouseup",stop);
      });
    });
  }

      var qtySpan = q.children[1];
      var plusBtn = q.children[2];

      minusBtn.onclick = function(){

        if(localQty > 0){

          localQty--;
          qtySpan.textContent = localQty;

          if(localQty === 0){
            remove(p.id,p.variants[0].id);
          }else{
            change(p.id,p.variants[0].id,-1);
          }
        }
      };

      plusBtn.onclick = function(){

        localQty++;
        qtySpan.textContent = localQty;

        if(localQty === 1){
          add(p,p.variants[0],1);
        }else{
          change(p.id,p.variants[0].id,1);
        }
      };

      e.appendChild(d);
    });
  }

  function syncCatalogQuantities(){

    var cards = document.querySelectorAll(".spcard");

    cards.forEach(function(card){

      var productId =
        card.getAttribute("data-product-id");

      var productData =
        P.find(function(p){
          return p.id === productId;
        });

      if(!productData)return;

      var defaultVariant = productData.variants[0];

      var item = cart.find(function(i){
        return i.productId === productId &&
               i.variantId === defaultVariant.id;
      });

      var span = card.querySelector(".spq span");

      if(span){
        span.textContent = item ? item.quantity : 0;
      }
    });
  }

  function renderCart(){

    var e = document.getElementById("spitems");
    if(!e)return;

    e.innerHTML = "";

    if(!cart.length){

      e.innerHTML =
        "<div class='spempty'>" +
          "<svg class='spempty-icon' viewBox='0 0 64 64' aria-hidden='true'>" +
            "<path d='M8 12h7l5 28h30l6-21H19' fill='none' stroke='currentColor' stroke-width='3' stroke-linecap='round' stroke-linejoin='round'/>" +
            "<circle cx='27' cy='52' r='3' fill='currentColor'/>" +
            "<circle cx='47' cy='52' r='3' fill='currentColor'/>" +
          "</svg>" +
          "<div class='spempty-title'>Tu carrito está vacío</div>" +
          "<div class='spempty-text'>Agrega productos para verlos aquí.</div>" +
        "</div>";

      syncCatalogQuantities();
      return;
    }

    cart.forEach(function(i){

      var d = document.createElement("div");
      d.className = "spi";

      d.innerHTML =
        "<span>" +
        i.name +
        " — " +
        money(i.price * i.quantity) +
        "</span>";

      var q = document.createElement("div");
      q.className = "spq";

      q.innerHTML =
        "<button type='button'>−</button>" +
        "<span>" + i.quantity + "</span>" +
        "<button type='button'>+</button>";

      q.children[0].onclick = function(){
        change(i.productId,i.variantId,-1);
      };

      q.children[2].onclick = function(){
        change(i.productId,i.variantId,1);
      };

      d.appendChild(q);

      var r = document.createElement("button");
      r.textContent = "×";
      r.title = "Eliminar producto";

      r.onclick = function(){
        remove(i.productId,i.variantId);
      };

      d.appendChild(r);
      e.appendChild(d);
    });

    var t = document.createElement("div");

    t.innerHTML =
      "<strong>Total: " +
      money(total()) +
      "</strong>";

    e.appendChild(t);

    syncCatalogQuantities();
  }

  function openCatalog(){

    catalog();
    renderCart();

    var windowCart=document.getElementById("spp");
    windowCart.style.left="50%";
    windowCart.style.top="50%";
    windowCart.style.transform="translate(-50%,-50%)";
    windowCart.classList.add("on");
    document.getElementById("spoverlay").classList.add("on");
  }

  function closeCatalog(){

    document.getElementById("spp").classList.remove("on");
    document.getElementById("spoverlay").classList.remove("on");
  }

  function openProduct(p){

    product = p;
    variant = p.variants[0];
    qty = 1;

    renderProduct();

    document.getElementById("spm").classList.add("on");
  }

  function closeProduct(){

    document.getElementById("spm").classList.remove("on");
  }

  function renderProduct(){

    var m = document.getElementById("spm");

    m.innerHTML =
      "<h2>" + product.name + "</h2>" +
      "<img src='" + product.image + "'>" +
      "<p>" + product.description + "</p>";

    var v = document.createElement("select");

    product.variants.forEach(function(vr){

      var o = document.createElement("option");

      o.value = vr.id;
      o.textContent =
        vr.name + " — " + money(vr.price);

      v.appendChild(o);
    });

    v.onchange = function(){

      variant =
        product.variants.find(function(x){
          return x.id === this.value;
        }.bind(this));
    };

    m.appendChild(v);

    var q = document.createElement("div");
    q.className = "spq";

    q.innerHTML =
      "<button id='menos'>−</button>" +
      "<span>" + qty + "</span>" +
      "<button id='mas'>+</button>";

    m.appendChild(q);

    m.querySelector("#menos").onclick = function(){

      if(qty > 1){
        qty--;
        renderProduct();
      }
    };

    m.querySelector("#mas").onclick = function(){
      qty++;
      renderProduct();
    };

    var addBtn = document.createElement("button");
    addBtn.className = "add";
    addBtn.textContent = "Agregar al carrito";

    addBtn.onclick = function(){

      add(product,variant,qty);
      closeProduct();
    };

    m.appendChild(addBtn);

    var buyBtn = document.createElement("button");
    buyBtn.className = "buy";
    buyBtn.textContent = "Comprar ahora";

    buyBtn.onclick = function(){
      window.open(product.url,"_blank");
    };

    m.appendChild(buyBtn);

    var closeBtn = document.createElement("button");
    closeBtn.className = "close-product";
    closeBtn.textContent = "Cerrar";
    closeBtn.onclick = closeProduct;

    m.appendChild(closeBtn);
  }

  function init(){

    css();

    var target = null;
    if (document.currentScript) {
      target = document.currentScript.getAttribute("data-target");
    }

    ui(target);
  }

  if(document.readyState === "loading"){
    document.addEventListener("DOMContentLoaded",init);
  }else{
    init();
  }

})();
